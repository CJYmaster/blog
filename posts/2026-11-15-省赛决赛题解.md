```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
C1 - "Baby RSA" (入门) 出题脚本

考点: RSA 小公钥指数 e=3 且无填充, 明文立方小于模数 n,
直接对密文开整数立方根即可恢复明文。

说明: 本脚本仅用于生成 output.txt。flag 与随机种子一律由环境变量注入
(EGEN_FLAG / EGEN_SEED), 源码中不保存任何明文答案, 也不使用固定种子,
避免附件本身泄漏答案 (明文 flag 或可复现的 p、q)。
"""
import os
import random

# 不内置真实 flag: 未注入时用一个明显的占位串（故意不用标准 flag 形状）
FLAG = os.environ.get("EGEN_FLAG", "PLACEHOLDER-SET-EGEN_FLAG-TO-GENERATE").encode()

# 不使用固定种子: 未注入 EGEN_SEED 时取系统随机数, 保证 p、q 不可复现
_seed = os.environ.get("EGEN_SEED")
random.seed(int(_seed, 0) if _seed else int.from_bytes(os.urandom(32), "big"))


def gen_prime(bits):
    while True:
        p = random.getrandbits(bits) | (1 << (bits - 1)) | 1
        if pow(2, p - 1, p) == 1 and pow(3, p - 1, p) == 1:
            return p


def bytes_to_long(b):
    return int.from_bytes(b, "big")


def long_to_bytes(n):
    return n.to_bytes((n.bit_length() + 7) // 8, "big")


# 明文 32 字节 = 256bit, m^3 约 768bit, 因此 n 需要 > 768bit
p = gen_prime(512)
q = gen_prime(512)
n = p * q
e = 3
m = bytes_to_long(FLAG)
assert m ** 3 < n, "m^3 应小于 n (立方根攻击条件)"
c = pow(m, e, n)

OUT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "attachment", "output.txt")
os.makedirs(os.path.dirname(OUT), exist_ok=True)
with open(OUT, "w") as f:
    f.write("n = %d\ne = %d\nc = %d\n" % (n, e, c))

print("[+] generated output.txt")
print("[+] check: m^3 < n ?", m ** 3 < n)
```

```python
import numpy as np
from secret import FLAG

MOD = 251

def get_affine_sbox():
    return lambda x: (7 * x + 123) % MOD

def mix_columns(state):
    matrix = np.array([[2, 3, 1, 1],
                       [1, 2, 3, 1],
                       [1, 1, 2, 3],
                       [3, 1, 1, 2]])
    return np.dot(matrix, state) % MOD

def encrypt(msg_bytes):
    msg = list(msg_bytes)
    while len(msg) % 16 != 0: msg.append(0)
    
    cipher_blocks = []
    sbox = get_affine_sbox()
    
    for i in range(0, len(msg), 16):
        block = np.array(msg[i:i+16]).reshape(4, 4)
        for _ in range(3):
            block = (7 * block + 123) % MOD
            for r in range(4):
                block[r] = np.roll(block[r], -r)
            block = mix_columns(block)
        cipher_blocks.append(block.flatten())
    return cipher_blocks

ct = encrypt(FLAG)
print(f"Ciphertext: {ct}")
```
### 题解
这是一个“类 AES”分组加密
整个加密过程是完全确定且可逆的。只要实现逆过程，就能从密文直接还原明文
#### 加密流程
S盒为仿射变换`sbox(x) = (7x + 123) mod 251`

`np.roll` 是 NumPy 的数组循环移位（滚动）函数
`1np.roll(a, shift, axis=None)`
a为待处理数组，shift为移位步数  正数向右或下滚动，负数为向左或上滚动
axis是在哪一维做移位，不指定时数组展平为1维滚动
题中第r行循环左移r位

mix_columns是点乘一个固定矩阵
#### 求逆步骤
1. 求固定矩阵模251的逆矩阵
2. 右移r位
3. S盒逆仿射`x = (y - 123) * 7^{-1} mod 251`

#### 解题代码
```python
import numpy as np  
from sympy import Matrix  
blocks = [np.array([ 19, 249, 206,  53, 181, 137, 230, 195, 146,  72, 135,  83,   7,  
        83, 141, 248]), np.array([137, 128, 109,  70,  27, 208, 126, 200, 214, 217, 235,  66,  44,  
       188, 227,  31]), np.array([221, 138,  45, 205,  29, 116,  98, 179, 182, 129, 103, 221,  45,  
        18,  25,  76])]  
MOD=251  
  
matrix = np.array([[2, 3, 1, 1],  
                       [1, 2, 3, 1],  
                       [1, 1, 2, 3],  
                       [3, 1, 1, 2]])  
M = Matrix(matrix)  
invM = M.inv_mod(MOD)  
invM=np.array(invM)  
  
def inv_shift_rows(block):  
    out = block.copy()  
    for r in range(4):  
        out[r] = np.roll(block[r], r)  
    return out
def inv_mix(block):  
    return (invM @ block) % MOD  
def inv_affine(block):  
    return ((block - 123) * pow(7,-1,MOD)) % MOD  
plain_blocks=[]  
for c in blocks:  
    block = c.reshape(4,4)  
    for _ in range(3):  
        block = inv_mix(block)  
        block = inv_shift_rows(block)  
        block = inv_affine(block)  
    plain_blocks.append(block.flatten())  
flat = []  
for b in plain_blocks:  
    flat += [int(x) for x in b]  
print(bytes(flat).rstrip(b'\x00'))
```
